project(AgenticNLE VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
option(NLE_SANITIZERS "Enable address and undefined behavior sanitizers on GCC/Clang" OFF)

option(NLE_BUILD_MCP "Build the optional local stdio MCP adapter" OFF)
if(NLE_BUILD_MCP)
  find_package(nlohmann_json 3.12.0 EXACT QUIET CONFIG)
  if(NOT nlohmann_json_FOUND)
    include(FetchContent)
    FetchContent_Declare(nlohmann_json
      URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
      URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    set(JSON_BuildTests OFF CACHE BOOL "Build JSON dependency tests" FORCE)
    FetchContent_MakeAvailable(nlohmann_json)
  endif()
  add_library(nle_mcp src/mcp/wire.cpp src/mcp/catalog.cpp src/mcp/session.cpp)
  target_link_libraries(nle_mcp PUBLIC nle_core nlohmann_json::nlohmann_json)
  add_executable(editor-mcp src/mcp/main.cpp)
  target_link_libraries(editor-mcp PRIVATE nle_mcp)
endif()
option(NLE_BUILD_DESKTOP "Build the optional Qt desktop and preview worker" OFF)
if(NLE_BUILD_DESKTOP)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network Multimedia Concurrent)
  add_library(nle_desktop src/desktop/transport.cpp src/desktop/transport.hpp src/desktop/window.cpp src/desktop/window.hpp)
  set_target_properties(nle_desktop PROPERTIES AUTOMOC ON)
  target_link_libraries(nle_desktop PUBLIC nle_playback nle_media Qt6::Widgets Qt6::Network Qt6::Concurrent)
  add_executable(nle-preview-worker src/desktop/worker.cpp)
  target_link_libraries(nle-preview-worker PRIVATE Qt6::Multimedia Qt6::Gui Qt6::Network)
  target_compile_features(nle-preview-worker PRIVATE cxx_std_20)
  add_executable(editor-desktop WIN32 src/desktop/main.cpp)
  target_link_libraries(editor-desktop PRIVATE nle_desktop)
  add_dependencies(editor-desktop nle-preview-worker)
  if(WIN32)
    get_target_property(NLE_QMAKE Qt6::qmake IMPORTED_LOCATION)
    get_filename_component(NLE_QT_BIN "${NLE_QMAKE}" DIRECTORY)
    add_custom_target(deploy-desktop
      COMMAND "${NLE_QT_BIN}/windeployqt.exe" --no-translations --no-compiler-runtime --include-plugins qoffscreen
        $<TARGET_FILE:editor-desktop> $<TARGET_FILE:nle-preview-worker>
      DEPENDS editor-desktop nle-preview-worker VERBATIM)
  endif()
endif()

  if(NLE_BUILD_MCP)

  add_executable(nle-source-timing-tests tests/source_timing.cpp)
  target_link_libraries(nle-source-timing-tests PRIVATE nle_media)

foreach(target nle-qt-recovery-tests nle-document-tests nle_mcp editor-mcp nle-mcp-tests nle-qt-precision-tests nle-source-timing-tests nle-hanging-worker editor-desktop nle-qt-desktop-tests nle_playback nle_desktop nle-preview-worker nle-playback-tests nle-qt-playback-tests nle_core nle_media nle-media-tests nle-process-helper editor-cli nle-tests nle-session-tests nle-benchmark)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
      if(NLE_WARNINGS_AS_ERRORS)
        target_compile_options(${target} PRIVATE /WX)
      endif()
    else()
      if(NLE_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
      endif()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
      if(NLE_WARNINGS_AS_ERRORS)
        target_compile_options(${target} PRIVATE -Werror)
      endif()
    endif()
  endif()
endforeach()
