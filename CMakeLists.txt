cmake_minimum_required(VERSION 3.24)
project(AgenticNLE VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
option(NLE_SANITIZERS "Enable address and undefined behavior sanitizers on GCC/Clang" OFF)
option(NLE_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)
add_library(nle_core src/core/time.cpp src/project/model.cpp src/commands/editor.cpp src/project/persistence.cpp)
target_include_directories(nle_core PUBLIC src)
target_compile_features(nle_core PUBLIC cxx_std_20)
target_link_libraries(nle_core PUBLIC Threads::Threads)
set_target_properties(nle_core PROPERTIES CXX_EXTENSIONS OFF)
add_executable(editor-cli src/cli/main.cpp)
target_link_libraries(editor-cli PRIVATE nle_core)
option(NLE_BUILD_BENCHMARKS "Build the repeatable session benchmark" OFF)
if(NLE_BUILD_BENCHMARKS)
  add_executable(nle-benchmark benchmarks/sessions.cpp)
  target_link_libraries(nle-benchmark PRIVATE nle_core)
endif()
if(BUILD_TESTING)
  add_executable(nle-session-tests tests/sessions.cpp)
  target_link_libraries(nle-session-tests PRIVATE nle_core)
  foreach(suite transactions rollback revisions concurrency history-limits track-media migration fuzz-persistence save-failure)
    add_test(NAME session-${suite} COMMAND nle-session-tests ${suite})
  endforeach()
  add_executable(nle-tests tests/tests.cpp)
  target_link_libraries(nle-tests PRIVATE nle_core)
  foreach(suite time creation insert move trim split delete history persistence validation)
    add_test(NAME ${suite} COMMAND nle-tests ${suite})
  endforeach()
  add_test(NAME cli-session-demo COMMAND editor-cli session-demo ${CMAKE_CURRENT_BINARY_DIR}/session-demo.nle)
  add_test(NAME cli-demo COMMAND editor-cli demo ${CMAKE_CURRENT_BINARY_DIR}/demo.nle)
  add_test(NAME cli-inspect COMMAND editor-cli inspect ${CMAKE_CURRENT_BINARY_DIR}/demo.nle)
  set_tests_properties(cli-demo PROPERTIES FIXTURES_SETUP demo)
  set_tests_properties(cli-inspect PROPERTIES FIXTURES_REQUIRED demo PASS_REGULAR_EXPRESSION "clips=2")
endif()
foreach(target nle_core editor-cli nle-tests nle-session-tests nle-benchmark)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
      if(NLE_WARNINGS_AS_ERRORS)
        target_compile_options(${target} PRIVATE /WX)
      endif()
    else()
      if(NLE_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
      endif()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
      if(NLE_WARNINGS_AS_ERRORS)
        target_compile_options(${target} PRIVATE -Werror)
      endif()
    endif()
  endif()
endforeach()
find_program(CLANG_FORMAT clang-format)
if(CLANG_FORMAT)
  file(GLOB_RECURSE FORMAT_SOURCES CONFIGURE_DEPENDS src/*.hpp src/*.cpp tests/*.cpp benchmarks/*.cpp)
  add_custom_target(format COMMAND ${CLANG_FORMAT} -i ${FORMAT_SOURCES} VERBATIM)
  add_custom_target(format-check COMMAND ${CLANG_FORMAT} --dry-run --Werror ${FORMAT_SOURCES} VERBATIM)
endif()
